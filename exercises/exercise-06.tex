\section*{Problem 1: Scale Matching in the Dualised Standard Model
  \hfill $\star\star\star$ \normalsize(25 min, 25 marks)}
\addcontentsline{toc}{section}{Problem 1: Scale matching -- $v$ and
  $\Lambda_{\QCD}$}
\label{prob:scale-matching}

\noindent\textbf{Background.}
In the dualised Standard Model, the \textbf{duality scale}
$\Lambda_D$ is the energy at which the electric $\SU{2}$ gauge
theory confines and the magnetic (SM) description becomes weakly
coupled.  Below $\Lambda_D$, the SM gauge couplings run according
to the usual perturbative renormalisation group equations.  The
duality scale is a \textbf{free parameter} of the framework; the
estimate $\Lambda_D \sim 10^{11}\;\text{GeV}$ is a parametric
consistency check (Lecture~8, \S4.1).

This exercise relates two physical scales to the duality scale and
asks what each relation tests:
\begin{enumerate}
  \item The electroweak (Fermi) scale $v = 246\;\text{GeV}$,
    through the seesaw-like relation
    $v \sim \Lambda_D^2 / M_{\text{Pl}}$.
  \item The QCD confinement scale
    $\Lambda_{\QCD} \approx 200\;\text{MeV}$,
    through one-loop RG running of $\alphas$ between $M_Z$ and
    $\Lambda_D$.
\end{enumerate}

\noindent\textbf{Questions.}
\begin{enumerate}[label=(\alph*)]

    The one-loop beta function for $\SU{3}_c$ QCD is:
    \begin{equation}\label{eq:beta-coeff}
      \bzero \;=\; \frac{11\Nc - 2\Nf}{3}\,.
    \end{equation}
    \begin{enumerate}[label=(\roman*)]
      \item With $\Nc = 3$ colours and $\Nf = 6$ quark flavours active
        above the top mass threshold, compute $\bzero$.

      \item The one-loop running of the strong coupling is
        (the one-loop solution of the beta function of Lecture~1):
        \begin{equation}\label{eq:one-loop-running}
          \alphas(\mu)
          \;=\; \frac{\alphas(\mu_0)}{%
            1 + \dfrac{\bzero}{2\pi}\,
            \alphas(\mu_0)\,
            \ln\!\bigl(\mu / \mu_0\bigr)}\,.
        \end{equation}
        The QCD confinement scale $\Lambda_{\QCD}$ is defined by the
        condition $\alphas(\Lambda_{\QCD}) \to \infty$, which gives:
        \begin{equation}\label{eq:lambda-qcd-def}
          \Lambda_{\QCD}
          \;=\; \mu_0 \;\exp\!\Bigl(
            -\frac{2\pi}{\bzero\,\alphas(\mu_0)}
          \Bigr)\,.
        \end{equation}

      \;\sim\; \alpha_m(\Lambda_D)
      \;=\; \alphas(\Lambda_D)\,.
    \end{equation}
    Strong coupling in the electric theory
    ($\alpha_e(\Lambda_D) \gg 1$) corresponds to weak coupling in the
    magnetic theory ($\alphas(\Lambda_D) \ll 1$).

    \begin{enumerate}[label=(\roman*)]

      \item Compare your result with the measured value

\begin{proof}[\textbf{Solution to Problem 1}]

\textbf{(a) QCD beta function and $\Lambda_{\QCD}$ definition.}

\textit{(i) Beta function coefficient.}

With $\Nc = 3$ and $\Nf = 6$:
\[
  \bzero \;=\; \frac{11 \times 3 - 2 \times 6}{3}
  \;=\; \frac{33 - 12}{3}
  \;=\; \frac{21}{3}
  \;=\; 7\,.
\]
\[
  \boxed{\bzero = 7\;\;\text{for } \SU{3},\; \Nf = 6\,.}
  \quad\checkmark
\]

Since $\bzero > 0$, QCD is asymptotically free with 6 flavours.

\textit{(ii) Verification of $\Lambda_{\QCD}$ definition.}

From Eq.~\eqref{eq:one-loop-running}, $\alphas(\mu) \to \infty$
when the denominator vanishes:
\[
  1 + \frac{\bzero}{2\pi}\, \alphas(\mu_0)\,
  \ln\!\bigl(\mu/\mu_0\bigr) \;=\; 0\,.
\]
Solving for $\mu$:
\[
  \ln\!\bigl(\mu/\mu_0\bigr)
  \;=\; -\frac{2\pi}{\bzero\, \alphas(\mu_0)}\,,
\]
\[
  \mu \;=\; \mu_0\, \exp\!\Bigl(
    -\frac{2\pi}{\bzero\, \alphas(\mu_0)}
  \Bigr)
  \;\equiv\; \Lambda_{\QCD}\,.
  \quad\checkmark
\]

\textit{(iii) Dimensional analysis.}

$\Lambda_{\QCD} = \mu_0 \times \exp(\ldots)$.  The exponential is
dimensionless, so
$[\Lambda_{\QCD}] = [\mu_0] = [\text{energy}]$.
\quad$\checkmark$

\bigskip

\textbf{(b) $\Lambda_{\QCD}$ from the duality scale.}

\textit{(i) Electric beta function.}

For the electric $\SU{2}$ theory with $n_f^e = 12$ Weyl doublets
(six $p$ and six $\widetilde{p}$) and 1 Weyl adjoint:
\begin{align}
  \bzero^e
  &\;=\; \frac{11 \times 2}{3}
  \;-\; \frac{2}{3} \times 12 \times \frac{1}{2}
  \;-\; \frac{2}{3} \times 2
  \nonumber \\
  &\;=\; \frac{22}{3} - 4 - \frac{4}{3}
  \;=\; \frac{22 - 4}{3} - 4
  \;=\; 6 - 4 \;=\; 2\,.
\end{align}
Equivalently, $\bzero^e = 3N_c^e - \tfrac{2}{3}\Nf = 6 - 4 = 2$ for
$\Nf = 6$ Dirac flavours plus one adjoint Weyl fermion.

Since $\bzero^e > 0$, the electric theory is \emph{asymptotically
free}: it becomes strongly coupled in the IR (at $\Lambda_D$) and
weakly coupled in the UV.  This is precisely the requirement for
duality---the electric theory confines at $\Lambda_D$, and the
magnetic (SM) description takes over below this scale.

\begin{remark}[Consistency check]
  The electric theory confining at $\Lambda_D$ requires that the
  electric coupling $\alpha_e$ grows as the energy decreases toward
  $\Lambda_D$. With $b_0^e > 0$ (asymptotic freedom), $\alpha_e$
  increases in the IR, consistent with the duality picture.
\end{remark}

\textit{(ii) Numerical computation of $\Lambda_{\QCD}$.}

\textbf{Step 1: run $\alphas(M_Z)$ up to $\Lambda_D$.}
With $\alphas(M_Z) = 0.118$, $M_Z = 91.2\;\text{GeV}$ and $\bzero = 7$,
Eq.~\eqref{eq:one-loop-running} gives
\[
  \alphas(\Lambda_D)
  \;=\; \frac{0.118}{%
    1 + \dfrac{7}{2\pi} \times 0.118 \times
    \ln\!\bigl(10^{11}/91.2\bigr)}\,.
\]
The logarithm is
\[
  \ln\!\bigl(10^{11}/91.2\bigr)
  \;=\; \ln(1.0965 \times 10^{9})
  \;=\; 9 \ln 10 + \ln 1.0965
  \;\approx\; 20.72 + 0.09
  \;=\; 20.82\,,
\]
and
\[
  \frac{7 \times 0.118}{2\pi}
  \;=\; \frac{0.826}{6.283}
  \;\approx\; 0.1315\,,
  \qquad
  0.1315 \times 20.82 \;\approx\; 2.736\,,
\]
so
\[
  \alphas(\Lambda_D)
  \;\approx\; \frac{0.118}{1 + 2.736}
  \;=\; \frac{0.118}{3.736}
  \;\approx\; 0.0316\,.
\]
The magnetic $\SU{3}_c$ coupling is weak at the duality scale, as the
duality relation~\eqref{eq:duality-coupling} requires for a strongly
coupled electric theory.

\textbf{Step 2: the exponent.}
\[
  \frac{2\pi}{\bzero\, \alphas(\Lambda_D)}
  \;=\; \frac{6.283}{7 \times 0.0316}
  \;=\; \frac{6.283}{0.2211}
  \;\approx\; 28.4\,.
\]

\textbf{Step 3: $\Lambda_{\QCD}$.}
\[
  \Lambda_{\QCD}
  \;=\; 10^{11}\;\text{GeV} \times \exp(-28.4)
  \;\approx\; 10^{11} \times 4.5 \times 10^{-13}\;\text{GeV}
  \;=\; 45\;\text{MeV}\,.
\]
\[
  \boxed{\Lambda_{\QCD}^{(\Nf=6)}
  \;\approx\; 45\;\text{MeV}\,.}
\]
The exponent needs three significant figures: rounding
$\alphas(\Lambda_D)$ to $0.032$ and the exponent to $28.0$ would give
$69\;\text{MeV}$, a $50\%$ error from percent-level roundings.

\textbf{Step 4: the same number from $M_Z$.}
\begin{align*}
  \Lambda_{\QCD}
  &\;=\; M_Z \exp\!\Bigl(-\frac{2\pi}{7 \times 0.118}\Bigr)
  \;=\; 91.2\;\text{GeV} \times \exp(-7.607) \\
  &\;\approx\; 91.2 \times 4.97 \times 10^{-4}\;\text{GeV}
  \;=\; 45\;\text{MeV}\,.
\end{align*}
The two results agree because the one-loop $\Lambda_{\QCD}$ is
renormalisation-group invariant.  From Eq.~\eqref{eq:one-loop-running},
$1/\alphas(\mu) = 1/\alphas(\mu_0) + (\bzero/2\pi)\ln(\mu/\mu_0)$, so
\[
  \ln\Lambda_{\QCD}
  \;=\; \ln\mu - \frac{2\pi}{\bzero\,\alphas(\mu)}
  \;=\; \ln\mu_0 - \frac{2\pi}{\bzero\,\alphas(\mu_0)}
\]
for any $\mu$.  Running $\alphas(M_Z)$ up to $\Lambda_D$ and
evaluating Eq.~\eqref{eq:lambda-qcd-def} there reproduces the value
obtained directly at $M_Z$: at one loop $\Lambda_D$ cancels, and the
QCD scale is fixed by $\alphas(M_Z)$ alone.

\textit{(iii) Comparison and sources of the difference.}

The one-loop result, $45\;\text{MeV}$, lies a factor of about $4.4$
below $\Lambda_{\QCD} \approx 200\;\text{MeV}$.  Two sources account
for the difference, and neither involves the duality:
\begin{enumerate}
  \item \textbf{Flavour thresholds.}  The value $45\;\text{MeV}$ is the
    six-flavour parameter $\Lambda^{(6)}$.  The number usually quoted is
    the five-flavour $\Lambda^{(5)}_{\overline{\text{MS}}}$.  Decoupling
    the top quark and running $\alphas(M_Z)$ with $\Nf = 5$
    ($\bzero = 23/3$) gives $\Lambda^{(5)} \approx 88\;\text{MeV}$ at one
    loop; continuing through $m_b$ and $m_c$ gives
    $\Lambda^{(4)} \approx 120\;\text{MeV}$ and
    $\Lambda^{(3)} \approx 143\;\text{MeV}$.  Below each threshold
    $\bzero$ is larger, which slows the running and raises $\Lambda$.

  \item \textbf{Loop order and scheme.}  $\Lambda$ depends on the order
    at which the beta function is truncated.  The two-loop formula gives
    $\Lambda^{(5)} \approx 230\;\text{MeV}$ from the same
    $\alphas(M_Z) = 0.118$, and the PDG value
    $\Lambda^{(5)}_{\overline{\text{MS}}} \approx 210\;\text{MeV}$ comes
    from four-loop running.
\end{enumerate}
Two further remarks.  Above $\Lambda_S$ the quasi-SUSY partners
(gaugino and squarks) change the $\SU{3}_c$ coefficient to the SQCD
value $\bzero = 3\Nc - \Nf = 3$; for $\Lambda_S = 3\;\text{TeV}$ this
raises $\alphas(\Lambda_D)$ from $0.032$ to about $0.048$ and leaves
$\Lambda_{\QCD}$, which is fixed by $\alphas(M_Z)$, unchanged.  And if
$\alphas(\Lambda_D)$ were computed from the duality
relation~\eqref{eq:duality-coupling} without using $\alphas(M_Z)$,
$\Lambda_{\QCD}$ would become a genuine, and exponentially sensitive,
prediction: a $30\%$ shift in $\alphas(\Lambda_D)$ moves
$\Lambda_{\QCD}$ up by a factor of about $700$ or down by a factor of
about $2 \times 10^{5}$.

\bigskip

\textbf{(c) The Fermi scale from the duality scale.}

\textit{(i) Seesaw relation.}

From Eq.~\eqref{eq:scalar-mass}, the Higgs mass parameter is
$\mu^2 \sim \xi\, \Lambda_D^4 / M_{\text{Pl}}^2$.  The electroweak
VEV is $v = \mu/\sqrt{\lambda}$, where $\lambda$ is the Higgs
quartic coupling (an $\mathcal{O}(1)$ number).  Therefore:
\[
  v \;\sim\; \frac{\mu}{\sqrt{\lambda}}
  \;\sim\; \frac{1}{\sqrt{\lambda}}\,
  \sqrt{\xi}\,
  \frac{\Lambda_D^2}{M_{\text{Pl}}}\,.
\]
Absorbing the $\mathcal{O}(1)$ factors $\sqrt{\xi}/\sqrt{\lambda}$:
\[
  \boxed{v \;\sim\; \frac{\Lambda_D^2}{M_{\text{Pl}}}\,.}
\]

\textbf{Dimensional check:}
$[\Lambda_D^2/M_{\text{Pl}}] = [\text{GeV}]^2 / [\text{GeV}]
= [\text{GeV}]$.
\quad$\checkmark$

\textbf{Physical interpretation.}  This is a seesaw-like relation: the
electroweak hierarchy $v/M_{\text{Pl}} \sim 10^{-17}$ arises as the
\emph{square} of the intermediate hierarchy
$\Lambda_D/M_{\text{Pl}} \sim 10^{-8}$:
\[
  \frac{v}{M_{\text{Pl}}}
  \;\sim\; \biggl(\frac{\Lambda_D}{M_{\text{Pl}}}\biggr)^{\!2}\,.
\]

\textit{(ii) Numerical evaluation.}

\[
  v \;\sim\; \frac{(10^{11})^2}{1.22 \times 10^{19}}\;\text{GeV}
  \;=\; \frac{10^{22}}{1.22 \times 10^{19}}\;\text{GeV}
  \;=\; \frac{10^{3}}{1.22}\;\text{GeV}
  \;\approx\; 820\;\text{GeV}\,.
\]
\[
  \boxed{v \;\sim\; 820\;\text{GeV}\,.}
\]

\textbf{Comparison:} The measured value is
$v_{\text{measured}} = 246\;\text{GeV}$.  The estimate
$v \sim 820\;\text{GeV}$ differs by a factor of $\sim 3.3$,
which is within the $\mathcal{O}(1)$ uncertainty of the framework
(the coefficient $\sqrt{\xi/\lambda}$ absorbs this factor).

The parametric dependence $v \propto \Lambda_D^2$ is correct: if we
adjust $\Lambda_D$ downward by a factor of $\sim 1.8$, we get
$v = 246\;\text{GeV}$ exactly.  This is a parametric consistency
check, not a precision prediction.

\textit{(iii) Range of $\Lambda_D$ for $v \in [100, 1000]$~GeV.}

Inverting the seesaw relation:
\[
  \Lambda_D \;\sim\; \sqrt{v \cdot M_{\text{Pl}}}\,.
\]
For $v = 100\;\text{GeV}$:
\[
  \Lambda_D \;\sim\; \sqrt{100 \times 1.22 \times 10^{19}}
  \;=\; \sqrt{1.22 \times 10^{21}}
  \;\approx\; 3.5 \times 10^{10}\;\text{GeV}\,.
\]
For $v = 1000\;\text{GeV}$:
\[
  \Lambda_D \;\sim\; \sqrt{1000 \times 1.22 \times 10^{19}}
  \;=\; \sqrt{1.22 \times 10^{22}}
  \;\approx\; 1.1 \times 10^{11}\;\text{GeV}\,.
\]
\[
  \boxed{\Lambda_D \in
  [3.5 \times 10^{10},\; 1.1 \times 10^{11}]\;\text{GeV}
  \quad\text{for}\quad
  v \in [100,\; 1000]\;\text{GeV}\,.}
\]

A factor $\sqrt{10} \approx 3.2$ in $\Lambda_D$ covers one order of
magnitude in $v$, since $v \propto \Lambda_D^2$.

\textit{(iv) Dimensional consistency.}

\[
  \biggl[\frac{\Lambda_D^2}{M_{\text{Pl}}}\biggr]
  \;=\; \frac{[\text{energy}]^2}{[\text{energy}]}
  \;=\; [\text{energy}]\,.
  \quad\checkmark
\]

  \qquad
  \text{Outputs: } v,\, \Lambda_S \text{ (one seesaw)};\;
  \alphas(\Lambda_D),\, \Lambda_{\QCD} \text{ (running)}\,.}
\]

\textit{(ii) Energy hierarchy.}

With measured or typical values, the hierarchy is:
\begin{align*}
  \Lambda_{\QCD} \;\sim\; 200\;\text{MeV}
  &\;\ll\; v \;=\; 246\;\text{GeV}
  \;\ll\; \Lambda_S \;\sim\; \text{few TeV} \\
  &\;\ll\; \Lambda_D \;\sim\; 10^{11}\;\text{GeV}
  \;\ll\; M_{\text{Pl}} \;\sim\; 10^{19}\;\text{GeV}\,.
\end{align*}

\textbf{Inputs:}
\begin{itemize}
  \item $M_{\text{Pl}}$: fundamental gravitational scale (measured).
  \item $\Lambda_D$: duality scale (free parameter; part~(c)(iii) gives
    $3.5 \times 10^{10}$--$1.1 \times 10^{11}\;\text{GeV}$ for
    $v$ between $100$ and $1000\;\text{GeV}$, up to $\mathcal{O}(1)$
    coefficients).
  \item $\alphas(M_Z) = 0.118$: measured.
\end{itemize}

\textbf{Outputs (derived):}
\begin{itemize}
  \item $v \sim \Lambda_D^2/M_{\text{Pl}}$: electroweak scale (seesaw
    relation).
  \item $\Lambda_S$: of order $\Lambda_D^2/M_{\text{Pl}}$ (the same
    seesaw as $v$, with a different $\mathcal{O}(1)$ coefficient).  It
    sets the masses of the quasi-SUSY states and of the 17 dormant Higgs
    doublets, the scale Exercise Set~5 uses for the dormant doublets.
  \item $\alphas(\Lambda_D) \approx 0.032$: weak magnetic coupling at
    the duality scale.
  \item $\Lambda_{\QCD}$: fixed by $\alphas(M_Z)$; at one loop it does
    not depend on $\Lambda_D$.
\end{itemize}

\textit{(iii) Prediction or consistency check?}

The relation $v \sim \Lambda_D^2/M_{\text{Pl}}$ is a
\textbf{parametric consistency check}:
\begin{itemize}
  \item The duality scale $\Lambda_D$ is a free parameter.  Given
    $\Lambda_D$, the framework produces $v$, but $\Lambda_D$ can be
    adjusted to match any desired $v$.
  \item The non-trivial content is that one intermediate scale,
    $\Lambda_D \sim 10^{11}\;\text{GeV}$, combined with $M_{\text{Pl}}$
    places both $v$ and $\Lambda_S$ near the TeV range with
    $\mathcal{O}(1)$ coefficients.  This counts as one check, since the
    same seesaw sets both scales.
  \item The QCD scale adds no independent test at one loop:
    $\Lambda_{\QCD}$ follows from $\alphas(M_Z)$ whatever the value of
    $\Lambda_D$ (part~(b)(ii)).  The running up to $\Lambda_D$ checks
    only that the magnetic coupling stays perturbative there
    ($\alphas(\Lambda_D) \approx 0.03$--$0.05$).  A genuine test would
    require $\alphas(\Lambda_D)$ from the duality relation itself.
\end{itemize}
\[
  \boxed{\begin{gathered}
  \text{Parametric consistency check: one input } (\Lambda_D)
  \text{ sets } v \text{ and } \Lambda_S \text{ through one seesaw;}\\
  \Lambda_{\QCD} \text{ is fixed by } \alphas(M_Z)
  \text{ and checks only that } \alphas(\Lambda_D) \text{ is perturbative.}
  \end{gathered}}
\]

\end{proof}


\bigskip

\begin{center}
\rule{0.6\textwidth}{0.4pt}

\textit{End of Exercise Set 6.  This exercise tests the
scale-matching framework of the dualised Standard Model.  The
seesaw-like relation turns the input $\Lambda_D \sim 10^{11}\;\text{GeV}$
(with the known Planck mass) into $v \approx 820\;\text{GeV}$, an
$\mathcal{O}(1)$ factor from the measured $246\;\text{GeV}$, and the
same seesaw sets the quasi-SUSY scale $\Lambda_S$.  One-loop running of
$\alphas(M_Z) = 0.118$ gives $\Lambda_{\QCD}^{(6)} \approx 45\;\text{MeV}$,
independent of $\Lambda_D$; flavour thresholds and higher-loop running
bring the quoted value to about $200\;\text{MeV}$.  Both results are
parametric consistency checks, not first-principles predictions: the
free parameter $\Lambda_D$ and the $\mathcal{O}(1)$ coefficients from
Planck-scale operators provide the flexibility to match the observed
scales.}
\end{center}
